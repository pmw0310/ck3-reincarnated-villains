# Victoria 3 Crossgame Portriat (3836483821) 크로스 서브모드 전용 복장 판정 트리거
# - 왕/황제 등극 이후 및 대규모 결혼식·감기·모략 등 상황에서도 중세 복장·왕관으로 회귀하지 않고 V3 근현대 복장을 유지합니다.
# - 단, 실제 소집 부대 지휘·야전 기사 출정·마상시합 무력 종목(난전/결투/마상창시합)·용병단/성전기사단 영주인 경우에만 각 고유 문화의 중세 갑옷·투구를 착용합니다.

villains_v3_is_in_combat_or_war_armor_trigger = {
	OR = {
		# 1. 실제 소집된 야전 부대 지휘관 또는 출정 부대 소속 기사 (평시 궁정·자문회·단순 전시 후방 체류 시 갑옷 미착용)
		portrait_wear_armor_currently_in_army_trigger = yes
		# 2. 마상시합 중 실제 무력 격투 종목(난전·결투·마상창시합) 참가 중일 때만 갑옷 착용 (시 낭송·체스·경마 등 비무력 종목 제외)
		AND = {
			exists = this
			is_incapable = no
			is_imprisoned = no
			involved_activity ?= {
				has_activity_type = activity_tournament
				OR = {
					has_current_phase = tournament_phase_melee
					has_current_phase = tournament_phase_duel
					has_current_phase = tournament_phase_joust
				}
			}
			is_in_guest_subset = { name = qualified }
		}
		# 3. 용병단·군사 성전기사단·도적 야영지 수장 (고유 전생 주인공 제외)
		AND = {
			NOT = { villains_v3_is_named_reincarnated_character_trigger = yes }
			is_ruler = yes
			OR = {
				government_has_flag = government_is_mercenary
				government_has_flag = government_is_military_holy_order
				has_realm_law = camp_purpose_mercenaries
				has_realm_law = camp_purpose_brigands
			}
		}
	}
}

villains_v3_situational_clear_trigger = {
	exists = this
	is_adult = yes
	portrait_prisoner_trigger = no
	should_be_naked_trigger = no
	should_be_topless_portrait_trigger = no
	should_be_fully_naked_portrait_trigger = no
}

villains_v3_use_modern_outfit_trigger = {
	villains_v3_situational_clear_trigger = yes
	NOT = { villains_v3_is_in_combat_or_war_armor_trigger = yes }
	NOT = {
		AND = {
			faith = faith:zionism
			has_council_position = councillor_court_chaplain
		}
	}
}
