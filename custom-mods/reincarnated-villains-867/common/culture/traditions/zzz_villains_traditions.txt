# 1. Israeli Exclusive: Iron Wall Citadel Keepers (철벽 요새 수호자)
tradition_iron_wall_citadels = {
	category = regional

	layers = {
		0 = martial
		1 = mena
		4 = city.dds
	}

	is_shown = {
		culture_is_or_has_ancestor_trigger = { CULTURE = culture:israeli }
		NOR = {
			has_cultural_tradition = tradition_castle_keepers
			has_cultural_tradition = tradition_city_keepers
		}
	}

	can_pick = {
		custom_tooltip = {
			text = tradition_iron_wall_citadels_israeli_only_tt
			culture_is_or_has_ancestor_trigger = { CULTURE = culture:israeli }
		}
		custom_tooltip = {
			text = must_be_feudal_or_clan
			scope:character = {
				OR = {
					government_has_flag = government_is_feudal
					government_has_flag = government_is_clan
					government_has_flag = government_is_administrative
					government_has_flag = government_is_republic
				}
			}
		}
		custom_tooltip = {
			text = cannot_have_tradition_castle_keepers
			NOT = {
				culture_tradition:tradition_castle_keepers = { is_in_list = traits }
			}
		}
		custom_tooltip = {
			text = cannot_have_tradition_city_keepers
			NOT = {
				culture_tradition:tradition_city_keepers = { is_in_list = traits }
			}
		}
	}

	parameters = {
		unlock_maa_sayeret_commandos = yes
		castle_fortifications_increase = yes
		castle_grant_prestige = yes
		architect_trait_prestige = yes
		ai_more_likely_to_castle = yes
		ai_more_likely_to_fortify = yes
	}

	character_modifier = {
		controlled_province_advantage = 6
		defender_holding_advantage = 6
		heavy_infantry_damage_mult = 0.10
		castle_holding_build_speed = -0.20
		city_holding_build_speed = -0.15
	}

	province_modifier = {
		fort_level = 1
		garrison_size = 0.25
		hostile_county_attrition = 1
	}

	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					NOR = {
						culture_pillar:ethos_bellicose = { is_in_list = traits }
						culture_pillar:ethos_stoic = { is_in_list = traits }
						culture_pillar:ethos_communal = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = not_bellicose_stoic_or_communal_desc
				}
			}
			multiply = tradition_replacement_cost_if_relevant
		}
	}

	ai_will_do = {
		value = 1000
	}
}
