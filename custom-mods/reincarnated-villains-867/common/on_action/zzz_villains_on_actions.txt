villains_jiang_qing_tang_emperor_secret_lover_on_game_start = {
	effect = {
		character:reincarnated_jiang_qing_867 ?= {
			save_scope_as = jiang_qing
			character:han_3641 ?= {
				save_scope_as = tang_emperor
			}
			if = {
				limit = {
					is_alive = yes
					exists = scope:tang_emperor
					scope:tang_emperor = { is_alive = yes }
				}
				if = {
					limit = { NOT = { has_relation_lover = scope:tang_emperor } }
					set_relation_lover = {
						reason = lover_history
						target = scope:tang_emperor
					}
				}
				if = {
					limit = {
						NOT = {
							any_secret = {
								type = secret_lover
								secret_target = scope:tang_emperor
							}
						}
					}
					add_secret = {
						type = secret_lover
						target = scope:tang_emperor
					}
				}
				scope:tang_emperor = {
					if = {
						limit = {
							NOT = {
								any_secret = {
									type = secret_lover
									secret_target = scope:jiang_qing
								}
							}
						}
						add_secret = {
							type = secret_lover
							target = scope:jiang_qing
						}
					}
				}
			}
		}
	}
}

					government_is_japanese_trigger = yes
				}
				house ?= {
					if = {
						limit = {
							NOT = {
								AND = {

					}

		}
	}

villains_rasputin_incapable_immunity_on_action = {
	effect = {
		if = {
			limit = {
				scope:trait = trait:incapable
				has_character_modifier = rasputin_herbal_contraception_modifier
			}
			remove_trait = incapable
			if = {
				limit = { has_character_flag = had_event_harm_0002 }
				remove_character_flag = had_event_harm_0002
			}
			if = {
				limit = { has_character_flag = had_event_harm_0003 }
				remove_character_flag = had_event_harm_0003
			}
			if = {
				limit = { has_trait = withering_mind }
				add_trait_xp = {
					trait = withering_mind
					track = withering_mind
					value = -50
				}
			}
			if = {
				limit = { has_trait = fragile_bones }
				add_trait_xp = {
					trait = fragile_bones
					track = fragile_bones
					value = -50
				}
			}
		}
	}
}
