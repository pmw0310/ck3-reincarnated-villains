villains_sync_siloviki_despot_opinion_effect = {
	if = {
		limit = {
			is_alive = yes
			is_ruler = yes
			is_landed = yes
			culture ?= { has_cultural_parameter = siloviki_despot_capital_popular_opinion }
			has_realm_law = republican_despot_succession_law
		}
		if = {
			limit = { NOT = { has_character_modifier = siloviki_despot_managed_opinion_modifier } }
			add_character_modifier = siloviki_despot_managed_opinion_modifier
		}
	}
	else_if = {
		limit = {
			has_character_modifier = siloviki_despot_managed_opinion_modifier
		}
		remove_character_modifier = siloviki_despot_managed_opinion_modifier
	}
}

# 단일 시작 명목 정치체 작위($REP_TITLE$)의 유령 시체 홀더(is_alive = no) 복구 헬퍼
# 구버전 세이브에서 환호 승계법 교착으로 사망자가 계속 작위를 쥐고 있는 경우 생존 상위 주군(또는 현지 신임 영주)에게 영지를 이양하고 명목 작위를 해체합니다.
villains_repair_single_dead_polity_holder_effect = {
	title:$REP_TITLE$ ?= {
		if = {
			limit = {
				is_title_created = yes
				exists = holder
				holder = { is_alive = no }
			}
			holder = {
				save_temporary_scope_as = dead_polity_holder
			}
			if = {
				limit = {
					exists = scope:dead_polity_holder.liege
					scope:dead_polity_holder.liege = {
						this != scope:dead_polity_holder
						is_alive = yes
					}
				}
				scope:dead_polity_holder.liege = {
					save_temporary_scope_as = escheat_receiver
				}
			}
			else_if = {
				limit = {
					exists = title:$FALLBACK_LIEGE$.holder
					title:$FALLBACK_LIEGE$.holder = {
						this != scope:dead_polity_holder
						is_alive = yes
					}
				}
				title:$FALLBACK_LIEGE$.holder = {
					save_temporary_scope_as = escheat_receiver
				}
			}
			else_if = {
				limit = {
					exists = scope:dead_polity_holder.capital_province
				}
				create_character = {
					location = scope:dead_polity_holder.capital_province
					culture = scope:dead_polity_holder.capital_county.culture
					rite = scope:dead_polity_holder.capital_county.rite
					gender_female_chance = 0
					age = { 24 42 }
					save_temporary_scope_as = escheat_receiver
				}
			}
			if = {
				limit = { exists = scope:escheat_receiver }
				create_title_and_vassal_change = {
					type = returned
					save_scope_as = corpse_cleanup_change
					add_claim_on_loss = no
				}
				scope:dead_polity_holder = {
					every_vassal = {
						limit = { is_alive = yes }
						change_liege = {
							liege = scope:escheat_receiver
							change = scope:corpse_cleanup_change
						}
					}
					every_held_title = {
						limit = {
							tier >= tier_county
							is_landless_type_title = no
							is_noble_family_title = no
							is_head_of_faith = no
							this != title:$REP_TITLE$
						}
						change_title_holder = {
							holder = scope:escheat_receiver
							change = scope:corpse_cleanup_change
							take_baronies = yes
						}
					}
					every_held_title = {
						limit = {
							OR = {
								this = title:$REP_TITLE$
								is_noble_family_title = yes
								is_head_of_faith = yes
							}
						}
						change_title_holder = {
							holder = scope:escheat_receiver
							change = scope:corpse_cleanup_change
						}
					}
				}
				resolve_title_and_vassal_change = scope:corpse_cleanup_change
				scope:escheat_receiver = {
					if = {
						limit = { has_title = title:$REP_TITLE$ }
						destroy_title = title:$REP_TITLE$
					}
					every_held_title = {
						limit = {
							is_noble_family_title = yes
							NOT = { this ?= scope:escheat_receiver.house.house_head.primary_title }
							NOR = {
								scope:escheat_receiver = { government_allows = noble_families }
								previous_holder ?= scope:escheat_receiver
							}
						}
						save_temporary_scope_as = stale_nf_title
						scope:escheat_receiver = { destroy_title = scope:stale_nf_title }
					}
					every_held_title = {
						limit = {
							is_head_of_faith = yes
							exists = controlled_faith
							controlled_faith != scope:escheat_receiver.faith
						}
						save_temporary_scope_as = stale_hof_title
						scope:escheat_receiver = { destroy_title = scope:stale_hof_title }
					}
				}
			}
		}
	}
}
