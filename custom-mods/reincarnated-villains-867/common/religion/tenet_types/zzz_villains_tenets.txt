# 1. Shared Ideology Tenet: Mass Propaganda & Mobilization
tenet_mass_propaganda = {
	icon = tenet_mass_propaganda

	name = {
		desc = tenet_mass_propaganda_name
	}

	desc = {
		desc = tenet_mass_propaganda_desc
	}

	historical_glossary = {
		first_valid = {
			desc = tenet_mass_propaganda_historical_glossary
		}
	}

	is_shown = {
		OR = {
			has_doctrine = special_doctrine_communist_ideology
			has_doctrine = special_doctrine_fascist_ideology
			has_doctrine = special_doctrine_capitalist_ideology
		}
	}

	can_pick = {
		custom_description = {
			text = requires_any_ideology_special_doctrine_trigger
			OR = {
				has_doctrine = special_doctrine_communist_ideology
				has_doctrine = special_doctrine_fascist_ideology
				has_doctrine = special_doctrine_capitalist_ideology
				doctrine:special_doctrine_communist_ideology = { is_in_list = selected_doctrines }
				doctrine:special_doctrine_fascist_ideology = { is_in_list = selected_doctrines }
				doctrine:special_doctrine_capitalist_ideology = { is_in_list = selected_doctrines }
			}
		}
	}

	can_pick_as_personal_tenet = {
		custom_description = {
			text = requires_any_ideology_special_doctrine_trigger
			faith = {
				OR = {
					has_doctrine = special_doctrine_communist_ideology
					has_doctrine = special_doctrine_fascist_ideology
					has_doctrine = special_doctrine_capitalist_ideology
				}
			}
		}
	}

	piety_cost = {
		value = faith_tenet_cost_low
		if = {
			limit = { rite_has_tenet = tenet_mass_propaganda }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	character_modifier = {
		monthly_legitimacy_add = 0.15
		diplomacy_per_piety_level = 1
	}

	county_modifier = {
		county_opinion_add = 10
		levy_reinforcement_rate = 0.2
	}

	parameters = {
		same_culture_conversion_bonus_active
		sway_scheme_piety
	}

	personal_tenet_modifier = {
		monthly_prestige_gain_mult = 0.15
		vassal_opinion = 5
		monthly_diplomacy_lifestyle_xp_gain_mult = 0.15
	}

	personal_tenet_parameters = {
		tenet_harmonious_society_sway_fulfillment
	}

	traits = {
		virtues = {
			gregarious = { weight = 15 scale = 1 }
			zealous = { weight = 15 scale = 1 }
		}
		sins = {
			shy = { weight = 15 scale = 1 }
			cynical = { weight = 15 scale = 1 }
		}
	}
}
