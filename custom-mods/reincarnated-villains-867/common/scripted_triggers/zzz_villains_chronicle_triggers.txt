# 연대기 '패권을 쥔 전생 왕조' 등급 판정:
# 장칭의 당나라 봉신용 시작 명목 왕국 직위(k_rep_jiang_qing)만 보유한 경우는 아직 상하이 3백작령 출발 단계이므로 제외하고,
# 실제로 상하이 인민코뮌(k_shanghai_commune) 등 정식 왕국·제국·패권국 이상을 수립/획득한 통치자만 패권 왕조로 분류합니다.
villains_is_ascendant_chronicle_ruler_trigger = {
	highest_held_title_tier >= tier_kingdom
	any_held_title = {
		tier >= tier_kingdom
		this != title:k_rep_jiang_qing
	}
}
